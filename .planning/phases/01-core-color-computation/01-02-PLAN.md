1. Add `scoreToHue()` as a module-level function above `updateMasteryColors()` (after line ~593, before the existing `updateMasteryColors` export). This is a pure function, no DOM access:

```typescript
function scoreToHue(score: number): number {
  if (score < 0) {
    if (score >= -5) {
      const t = Math.sqrt((-score) / 5);
      return 90 + t * (264 - 90);
    }
    const t = Math.sqrt((-score - 5) / 5);
    return 27 + t * (90 - 27);
  }
  const t = Math.sqrt(score / 10);
  return 264 - t * (264 - 145);
}
```

Hue anchors: red=27, yellow=90, blue=264, green=145. Sqrt easing applied to the segment-local `t` parameter (NOT to the hue angle). Direction: score -1 is near blue (264), score -5 is near yellow (90), score -10 is near red (27), score +10 is near green (145).

2. Add `getMasteryConstants()` right after `scoreToHue`. Reads the four CSS custom properties once (NOT per-cell):

```typescript
interface MasteryConstants { bgL: string; bgC: string; fgL: string; fgC: string; }

function getMasteryConstants(): MasteryConstants {
  const s = getComputedStyle(document.documentElement);
  return {
    bgL: s.getPropertyValue('--mastery-bg-L').trim(),
    bgC: s.getPropertyValue('--mastery-bg-C').trim(),
    fgL: s.getPropertyValue('--mastery-fg-L').trim(),
    fgC: s.getPropertyValue('--mastery-fg-C').trim(),
  };
}
```

3. Rewrite the body of `updateMasteryColors()` (keep the export, keep the function signature). Replace the entire body with:

```typescript
export function updateMasteryColors(): void {
  if (!appState.keys.length) return;
  const { bgL, bgC, fgL, fgC } = getMasteryConstants();
  const cells = document.querySelectorAll('#grid .grid-cell');
  cells.forEach((cell) => {
    const key = cell.getAttribute('data-key');
    if (!key) return;
    const raw = (MODES.quiz.scores[key] ?? 0)
              + (MODES.reverse.scores[key] ?? 0)
              + (MODES.mixed.scores[key] ?? 0);
    const score = Math.max(-10, Math.min(10, raw));
    const el = cell as HTMLElement;
    el.className = el.className.replace(/mastery-\d/g, '').trim();
    if (score === 0) {
      el.style.backgroundColor = '';
      el.style.color = '';
      return;
    }
    const H = scoreToHue(score);
    el.style.backgroundColor = `oklch(${bgL} ${bgC} ${H})`;
    el.style.color = `oklch(${fgL} ${fgC} ${H})`;
  });
}
```

Key differences from old code:
- Reads CSS custom props via getMasteryConstants() instead of parseColor(--bg-surface)
- Sets BOTH `el.style.backgroundColor` AND `el.style.color` (old code only set backgroundColor)
- Score 0 now clears BOTH inline styles (old code only cleared backgroundColor)
- Uses oklch() template literal instead of rgb() from lerpRGB

Do NOT remove: `parseColor`, `lerpRGB`, `GRADIENT_YELLOW`, `GRADIENT_RED`, `GRADIENT_GREEN`, `type RGB`. These are cleaned up in Phase 3.
Do NOT remove: the `el.className.replace(/mastery-\d/g, '')` line. That is also Phase 3 cleanup.
